- T2. runnable as tool

```python
from langchain_core.runnables import RunnableLambda
runnable = RunnableLambda(lambda x: x["a"] + x["b"])
print(runnable.invoke({"a": 2, "b": 3}))

add = runnable.as_tool(name="add", description="Add two numbers.", arg_types={"a": int, "b": int})
```

- T3. 영어를 한국어로 번역하는 도구

```python
from langchain_core.prompts import PromptTemplate
prompt = PromptTemplate.from_template("""
다음 영어 문장을 한국어로 번역해 주세요.\nEnglish: {text}
""")

from langchain_google_genai import ChatGoogleGenerativeAI
llm = ChatGoogleGenerativeAI(model="gemini-robotics-er-1.5-preview")
chain = prompt | llm

translator = chain.as_tool(name="translate", description="주어진 영어 문장을 한국어로 번역.", arg_types={"text": str})

response = translator.invoke({"text": "Hello, how are you?"})
print(response.content)
```

- T4. Tool로

```python
from langchain_core.tools import tool
from datetime import datetime

@tool
def get_today():
    """Get today's date in the format 'YYYY년 MM월 DD일 A'."""
    return datetime.today().strftime("%Y년 %m월 %d일 %A")

from langchain_google_genai import ChatGoogleGenerativeAI
llm = ChatGoogleGenerativeAI(model="gemini-robotics-er-1.5-preview")
llm_with_tools = llm.bind_tools([get_today])

from langchain_core.messages import SystemMessage, HumanMessage, ToolMessage
# SystemMessage를 제공하지 않으면 tool을 사용하지 않을 수도 있다.
# 어떤 도구를 언제 사용해야 하는지를 명확히 하는 것이 좋다.
messages = [SystemMessage(content="오늘의 날짜와 요일은 get_today 도구를 사용해서 확인합니다."),
            HumanMessage(content="모레는 무슨 요일인가요?")]

ai_message = llm_with_tools.invoke(messages)
print(ai_message)
messages.append(ai_message)

for tool_call in ai_message.tool_calls:
    if tool_call["name"] == "get_today":
#        messages.append(get_today.invoke(tool_call))
        messages.append(ToolMessage(content=get_today.invoke({}),
                                    tool_call_id=tool_call["id"]))

ai_message = llm_with_tools.invoke(messages)
print(ai_message)
```

- T5. create_tool_calling_agent

```python
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
prompt = ChatPromptTemplate.from_messages([
    ("system", "당신은 도구를 사용하는 AI입니다. 오늘 날짜와 요일은 get_today 도구를 사용해서 확인합니다."),
    ("user", "{input}"),
    MessagesPlaceholder(variable_name="agent_scratchpad"),  # tool과의 대화를 저장한다.
])

from langchain.agents import create_tool_calling_agent, AgentExecutor
agent = create_tool_calling_agent(llm, [get_today], prompt)
agent_executor = AgentExecutor(agent=agent, tools=[get_today])
response = agent_executor.invoke("모레는 무슨 요일인가요?")
print(response)
```

## P3-6

- T1. MCP Client

  - sync

  ```python
  def func(message):
      print(message)

  def main():
      func("Hello")
      func("World")

  main()
  ```

  - async

  ```python
  import asyncio

  async def func(delay, message):
      print(message)
      await asyncio.sleep(delay)
      print(message)

  async def main():
      f1 = func(2, "Hello")   # coroutine object f1
      f2 = func(1, "World")   # coroutine object f2
      await asyncio.gather(f1, f2)

  asyncio.run(main())
  ```

  - create_tool_calling_agent()를 사용해서 MCP tools을 사용하도록 할 수 있다.

  ```python
  import asyncio

  async def main():
      tools = await client.get_tools()    # asynchronous call to get tools

      from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
      from langchain.agents import create_tool_calling_agent, AgentExecutor
      from langchain_google_genai import ChatGoogleGenerativeAI

      llm = ChatGoogleGenerativeAI(model="gemini-robotics-er-1.5-preview")
      prompt = ChatPromptTemplate.from_messages([
          ("system", "당신은 수학 문제를 푸는 도우미입니다. 도구를 사용해서 문제를 해결하세요."),
          ("user", "{input}"),
          MessagesPlaceholder(variable_name="agent_scratchpad"),
      ])
      agent = create_tool_calling_agent(llm, tools, prompt)
      agent_executor = AgentExecutor(agent=agent, tools=tools)

      response = await agent_executor.ainvoke({"input": "2와 3을 더한 결과에 5를 곱하면 얼마인가요?"})
      print(response)

  if __name__ == "__main__":
      asyncio.run(main())
  ```
